%% One-page classic resume template
%% Recreates the structure of a user-supplied Word template:
%% Name/contact header -> Education -> Organizational and Technical Experience
%% -> Technical Skills -> Leadership and Activities. Times New Roman, tight
%% margins, left-title/right-date subheadings. Hard 1-page limit.
%% Compile with: lualatex -interaction=nonstopmode template.tex

\documentclass[11pt,letterpaper]{article}

% TeX Gyre Termes is an OpenType Times New Roman clone bundled with every
% modern TeX distribution (MiKTeX/TeX Live). Loaded via fontspec under
% lualatex so bold/italic shapes resolve correctly - the classic `times`
% (PSNFSS) package leaves bold/italic undefined under lualatex's default
% TU/Unicode font encoding and silently substitutes regular weight instead.
\usepackage{fontspec}
\setmainfont{TeX Gyre Termes}
\usepackage[
  top=0.5in, bottom=0.4in, left=0.75in, right=0.5in
]{geometry}
\usepackage{titlesec}
\usepackage{enumitem}
\usepackage[hidelinks]{hyperref}

\pagestyle{empty}
\setlength{\tabcolsep}{0in}

% Section headers: bold, small caps look, full-width rule underneath
\titleformat{\section}{\raggedright\large\bfseries}{}{0em}{}[\titlerule\vspace{-4pt}]
\titlespacing*{\section}{0pt}{10pt}{6pt}

% Tight itemize for bullets
\setlist[itemize]{leftmargin=0.18in, itemsep=0pt, parsep=0pt, topsep=0pt, partopsep=0pt}

% Left title / right date subheading row.
% Uses an explicit tabular* (fixed two-cell row) rather than \hfill glue:
% poppler's `pdftotext -layout` (the ATS-parseability check's own extraction
% tool - see TEMPLATE.md "Known pitfalls") misreads \hfill-separated
% left/right text as a shifting multi-column layout once title lengths vary
% between entries, scrambling which date attaches to which entry in the
% extracted text. A real table row extracts in the correct left-to-right
% order every time.
% A tabular*/extracolsep row still let pdftotext -layout misattribute dates
% to the wrong entry: each entry is its own tabular*, so its right cell's
% start-x still varies with THAT row's title width, exactly like \hfill did.
% poppler's -layout heuristic clusters text into columns using a single
% column-boundary x-coordinate for the whole page; a boundary that shifts
% entry-to-entry breaks that model and it starts pairing a date with the
% wrong text run. The stock moderncv template (used elsewhere in this repo)
% avoids this because its date column sits at a FIXED x-offset on every
% entry. \makebox[<fixed width>][l/r]{...} reproduces that: the date box
% always starts at exactly \linewidth-\ppdatewidth, identical on every row
% regardless of title length, which keeps poppler's column detection stable
% and each date correctly paired with its own entry.
\newlength{\ppdatewidth}
\setlength{\ppdatewidth}{2.2in}
% \noindent is required on every call site: without it, any entry that
% doesn't immediately follow a \section (i.e. every entry after the first
% in a section) inherits \parindent, shifting that row's title right
% relative to sibling entries. Keep the 3rd/4th subtitle arguments SHORT
% (a role/subtitle line, not a full sentence) and the 1st/3rd (title) args
% under roughly \linewidth-\ppdatewidth wide: \makebox does not wrap, so
% overlong text overflows past its box instead of wrapping to a new line.
% A long description belongs as an ordinary line of text after the
% \subheading call, not packed into one of its arguments.
\newcommand{\subheading}[4]{
  \noindent\makebox[\dimexpr\linewidth-\ppdatewidth\relax][l]{\textbf{#1}}\makebox[\ppdatewidth][r]{#2}\\
  \makebox[\dimexpr\linewidth-\ppdatewidth\relax][l]{\textit{#3}}\makebox[\ppdatewidth][r]{\textit{#4}}\\[-2pt]
}

\newcommand{\subheadingnodate}[2]{
  \noindent\makebox[\dimexpr\linewidth-\ppdatewidth\relax][l]{\textbf{#1}}\makebox[\ppdatewidth][r]{#2}\\[-2pt]
}
